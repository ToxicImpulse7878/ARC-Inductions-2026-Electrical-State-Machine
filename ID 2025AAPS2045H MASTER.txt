// ID 2025AAPS2045H MASTER ARDUINO CODE

#include <Servo.h>
#include <Wire.h>

// PIN DEFINITIONS 
const int PIN_LDR         = A0;
const int PIN_GAS_SENSOR  = A1;
const int PIN_TEMP_SENSOR = A2;
const int PIN_SERVO       = 9;
const int PIN_BUZZER      = 8;
// A4 = SDA and A5 = SCL are used by the I2C bus. Never wire anything else
// to them and never ground them.

// I2C
const uint8_t SLAVE_ADDRESS      = 8;
const unsigned long I2C_INTERVAL = 250;  // ms between polls
// 250 rather than 50, found empirically. At 50 ms the master's polls landed
// inside the slave's LCD writes often enough to corrupt characters
// ("AWAITING RITUA", "BOOTö≷r"). Polling less often makes collisions rare.
// The display only redraws on change, so nothing is lost by the slower rate.
unsigned long lastI2CService     = 0;

// STATE CODES (must match the slave exactly) 
const uint8_t STANDBY           = 0;
const uint8_t ACTIVE_MONITORING = 1;
const uint8_t GAS_ALERT         = 2;
const uint8_t BLACKOUT_ALERT    = 3;
const uint8_t TEMP_EMERGENCY    = 4;
const uint8_t MULTI_FAULT       = 5;
const uint8_t TEMP_MULTI_FAULT  = 6;

uint8_t currentState  = STANDBY;
uint8_t previousState = 255;   // impossible value forces one log on first pass

// COMMAND CODES (must match the slave exactly)
const uint8_t CMD_NONE     = 0;
const uint8_t CMD_ACTIVATE = 1;
const uint8_t CMD_TOGGLE   = 2;
const uint8_t CMD_RESET    = 3;

uint8_t pendingIRCommand = CMD_NONE;

// FAULT AND CONTROL FLAGS
bool activated            = false;
bool showGasOnLCD         = false;   // false = light levels, true = air purity
bool gasFaultLatched      = false;
bool lightFaultLatched    = false;
bool tempFaultLatched     = false;
bool resetCommandReceived = false;

// SENSOR READINGS, refreshed once per loop pass
int   gasReading   = 0;     // raw ADC, 0..1023
int   lightReading = 0;     // raw ADC, 0..1023
float tempC        = 0.0;   // degrees Celsius, already converted

// THRESHOLDS
const int   GAS_TRIGGER_ON    = 180;   // raw ADC, from the spec
const int   GAS_TRIGGER_OFF   = 130;   // raw ADC, from the spec
// Calibrated against this circuit's LDR + 10k divider: 6 raw in full dark,
// 679 raw at full brightness. Thresholds sit well inside that span, with a
// wide gap between them so the blackout state cannot chatter at the boundary.
const int   LIGHT_TRIGGER_ON  = 150;   // raw ADC, calibrated
const int   LIGHT_TRIGGER_OFF = 450;   // raw ADC, calibrated
const float TEMP_TRIGGER_ON   = 45.0;  // degrees Celsius, from the spec

const int SERVO_RESTING = 0;
const int SERVO_VENTING = 180;
const int BUZZER_TONE_HZ = 1000;

Servo ventServo;


void setup() {
  Serial.begin(9600);

  pinMode(PIN_BUZZER, OUTPUT);
  noTone(PIN_BUZZER);

  ventServo.attach(PIN_SERVO);
  ventServo.write(SERVO_RESTING);

  Wire.begin();   // no argument = join the bus as MASTER

  Serial.println(F("Master ready."));
}

void loop() {
  // 1. INPUTS FIRST. Commands must be consumed before the flags that
  //    depend on them are evaluated, or a reset arrives one pass too late.
  checkIRCommands();

  // 2. UPDATE EVERY FAULT FLAG, EVERY PASS, NO EXCEPTIONS.
  updateGasFault();
  updateLightFault();
  updateTempFault();

  // 3. DECIDE.
  currentState = decideState();

  // 4. ACT on the master's own actuators.
  executeStateActions(currentState);

  // 5. LOG TRANSITIONS ONLY. Satisfies State 3's "logs the event", and
  //    keeps the serial monitor readable instead of thousands of
  //    identical lines per second.
  if (currentState != previousState) {
    Serial.print(F("State -> "));
    Serial.println(stateName(currentState));
    previousState = currentState;
  }

  // 6. I2C SERVICE, RATE LIMITED.
  if (millis() - lastI2CService >= I2C_INTERVAL) {
    lastI2CService = millis();
    serviceI2C();
  }
}

// FAULT DETECTION — each function is a self-contained hysteresis latch.

void updateGasFault() {
  gasReading = readGasSensor();

  if (gasReading > GAS_TRIGGER_ON) {
    gasFaultLatched = true;
  } else if (gasReading < GAS_TRIGGER_OFF) {
    gasFaultLatched = false;
  }
  // Between the two thresholds the latch holds its previous value. That
  // dead zone is the hysteresis band; it stops the flag chattering on a
  // reading hovering at the limit. It also implements the spec's
  // "stays in this state until gas drops below 130, returning
  // automatically to State 1".
}

void updateLightFault() {
  lightReading = readLightSensor();

  if (lightReading < LIGHT_TRIGGER_ON) {
    lightFaultLatched = true;
  } else if (lightReading > LIGHT_TRIGGER_OFF) {
    lightFaultLatched = false;
  }
}

void updateTempFault() {
  tempC = readTempSensor();

  if (tempC > TEMP_TRIGGER_ON) {
    tempFaultLatched = true;
  }
  // Deliberately no automatic clear. Per the spec, State 4 "requires a
  // manual reset command sent via the IR remote to exit this state".

  if (resetCommandReceived) {
    if (tempC <= TEMP_TRIGGER_ON) {
      tempFaultLatched = false;
      Serial.println(F("Temp latch cleared by reset."));
    } else {
      // Without this guard the latch would clear and immediately re-set
      // from the still-high reading on the next pass, swallowing the
      // command with no feedback.
      Serial.println(F("Reset refused: temperature still above threshold."));
    }
    resetCommandReceived = false;   // consumed either way, so a refused
                                    // reset cannot fire later on its own
  }
}

// STATE ARBITRATION — one linear priority chain, evaluated fresh each pass.

uint8_t decideState() {
  // Temperature plus at least one other fault. Extension beyond the spec —
  // see the note by TEMP_MULTI_FAULT. Placed above TEMP_EMERGENCY so the
  // compound case is reported rather than hidden behind it.
  if (tempFaultLatched && (gasFaultLatched || lightFaultLatched)) {
    return TEMP_MULTI_FAULT;
  }

  // State 4 overrides States 0-3, Standby included, so this sits above the
  // activation gate.
  if (tempFaultLatched) {
    return TEMP_EMERGENCY;
  }

  if (!activated) {
    return STANDBY;
  }

  if (gasFaultLatched && lightFaultLatched) {
    return MULTI_FAULT;
  } else if (gasFaultLatched) {
    return GAS_ALERT;
  } else if (lightFaultLatched) {
    return BLACKOUT_ALERT;
  } else {
    return ACTIVE_MONITORING;
  }
}

// ACTION EXECUTION — master-owned actuators only.

void executeStateActions(uint8_t state) {
  // Servo opens the vent during ANY temperature emergency, compound or not.
  // Critical: if TEMP_MULTI_FAULT were omitted here, the vent would close
  // exactly when conditions are worst — overheating plus a second fault.
  if (state == TEMP_EMERGENCY || state == TEMP_MULTI_FAULT) {
    ventServo.write(SERVO_VENTING);
  } else {
    ventServo.write(SERVO_RESTING);
  }

  // Buzzer sounds continuously only during a multi-fault.
  //
  // tone() rather than digitalWrite(HIGH): a piezo responds to an
  // alternating voltage, not a steady one. A constant HIGH produces a
  // single click and then silence. tone() uses Timer2 while the Servo
  // library uses Timer1, so the two do not conflict on an Uno.
  if (state == MULTI_FAULT || state == TEMP_MULTI_FAULT) {
    tone(PIN_BUZZER, BUZZER_TONE_HZ);
  } else {
    noTone(PIN_BUZZER);
  }
}

// SENSOR READING

int readGasSensor() {
  return analogRead(PIN_GAS_SENSOR);   // raw ADC; thresholds are also raw
}

int readLightSensor() {
  return analogRead(PIN_LDR);          // raw ADC; thresholds are also raw
}

float readTempSensor() {
  // The analog pin measures VOLTAGE, not heat, and reports it as 0..1023.
  // Comparing that count directly against 45 would latch a temperature
  // emergency at room temperature, because room temperature reads ~147.
  int raw     = analogRead(PIN_TEMP_SENSOR);
  float volts = raw * (5.0 / 1024.0);
  return (volts - 0.5) * 100.0;   // TMP36: 500 mV offset at 0 C, 10 mV per C
}

// IR COMMAND HANDLING

void checkIRCommands() {
  // State 3: "ignores standard IR commands until lighting is restored."
  //
  // Gated on the STATE rather than on lightFaultLatched. The latch can
  // already be set while in Standby, since the update functions run
  // unconditionally, and IR is the only way to activate the system —
  // gating on the flag would make a dark room a permanent lockout.
  if (currentState == BLACKOUT_ALERT || currentState == MULTI_FAULT) {
    pendingIRCommand = CMD_NONE;   // dropped, not queued: it must not fire
    return;                        // later once lighting is restored
  }

  if (pendingIRCommand == CMD_NONE) {
    return;
  }

  switch (pendingIRCommand) {
    case CMD_ACTIVATE: enterActive();               break;
    case CMD_TOGGLE:   toggleDisplayMode();         break;
    case CMD_RESET:    resetCommandReceived = true; break;
  }
  pendingIRCommand = CMD_NONE;
}

// Clears any latch set while the system was switched off, so monitoring
// starts from a clean slate rather than instantly alarming on a stale
// event. Matters most for temperature, which has no automatic clear and
// would otherwise carry a fault across the entire off period.
void enterActive() {
  if (activated) return;   // already on; ignore repeat presses, which
                           // would otherwise wipe live faults

  gasFaultLatched      = false;
  lightFaultLatched    = false;
  tempFaultLatched     = false;
  resetCommandReceived = false;

  activated = true;
  Serial.println(F("Activated — latches cleared, monitoring live."));
}

void toggleDisplayMode() {
  showGasOnLCD = !showGasOnLCD;
  Serial.print(F("LCD mode -> "));
  Serial.println(showGasOnLCD ? F("GAS") : F("LIGHT"));
}

// I2C SERVICE — called on a timer, never every pass.

void serviceI2C() {
  // (a) PULL the slave's pending IR command. The slave clears its slot
  //     only inside its onRequest handler, so a command cannot be lost to
  //     a poll that never happens.
  uint8_t received = CMD_NONE;
  if (Wire.requestFrom((uint8_t)SLAVE_ADDRESS, (uint8_t)1) == 1) {
    received = Wire.read();
  }
  if (received != CMD_NONE) {
    pendingIRCommand = received;
  }

  // (b) PUSH the display payload. Every value cast explicitly: Wire.write()
  //     is overloaded and a bare integer expression can compile as an
  //     ambiguous call.
  Wire.beginTransmission(SLAVE_ADDRESS);
  Wire.write((uint8_t)currentState);
  Wire.write((uint8_t)(showGasOnLCD ? 1 : 0));
  Wire.write((uint8_t)lightPercent());
  Wire.write((uint8_t)purityPercent());
  Wire.endTransmission();
}

// DISPLAY UNIT CONVERSIONS

uint8_t lightPercent() {
  long pct = map(lightReading, 0, 1023, 0, 100);
  return (uint8_t)constrain(pct, 0, 100);
}

// Air PURITY is the inverse of the gas reading: a high gas reading means
// dirty air. Passing the raw value straight through would show high purity
// during a gas alert.
uint8_t purityPercent() {
  long pct = 100 - map(gasReading, 0, 1023, 0, 100);
  return (uint8_t)constrain(pct, 0, 100);
}

// UTILITY

const char* stateName(uint8_t s) {
  switch (s) {
    case STANDBY:           return "STANDBY";
    case ACTIVE_MONITORING: return "ACTIVE_MONITORING";
    case GAS_ALERT:         return "GAS_ALERT";
    case BLACKOUT_ALERT:    return "BLACKOUT_ALERT";
    case TEMP_EMERGENCY:    return "TEMP_EMERGENCY";
    case MULTI_FAULT:       return "MULTI_FAULT";
    case TEMP_MULTI_FAULT:  return "TEMP_MULTI_FAULT";
    default:                return "UNKNOWN";
  }
}
