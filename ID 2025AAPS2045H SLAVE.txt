// ID 2025AAPS2045H SLAVE CODE

#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include <IRremote.h>

// CONFIGURATION 
const uint8_t SLAVE_ADDRESS = 8;
const uint8_t LCD_ADDRESS   = 0x20;   // Tinkercad shows this as 32
const int     PIN_IR        = 2;

LiquidCrystal_I2C lcd(LCD_ADDRESS, 16, 2);

// STATE CODES (must match the master exactly)
const uint8_t STANDBY           = 0;
const uint8_t ACTIVE_MONITORING = 1;
const uint8_t GAS_ALERT         = 2;
const uint8_t BLACKOUT_ALERT    = 3;
const uint8_t TEMP_EMERGENCY    = 4;
const uint8_t MULTI_FAULT       = 5;
const uint8_t TEMP_MULTI_FAULT  = 6;   // extension beyond the spec; see master

// COMMAND CODES (must match the master exactly)
const uint8_t CMD_NONE     = 0;
const uint8_t CMD_ACTIVATE = 1;
const uint8_t CMD_TOGGLE   = 2;
const uint8_t CMD_RESET    = 3;

// IR BUTTON MAPPING
// Captured from Tinkercad's remote with a probe sketch.
// The power button only ever activates; the spec defines no deactivate
// command, so pressing it again is ignored by the master.
const uint8_t IR_CODE_ACTIVATE = 0x00;   // power button
const uint8_t IR_CODE_TOGGLE   = 0x09;   // volume down
const uint8_t IR_CODE_RESET    = 0x0C;   // the "0" key

const unsigned long IR_DEBOUNCE_MS = 200;

volatile uint8_t pendingCommand = CMD_NONE;
volatile uint8_t rxBuffer[4]    = {STANDBY, 0, 0, 0};
volatile bool    newDataReady   = false;

// LAST RENDERED PAYLOAD (change detection)
uint8_t lastState = 255;   // impossible value forces a render on first pass
uint8_t lastMode  = 255;
uint8_t lastLight = 255;
uint8_t lastGas   = 255;

unsigned long lastIRAccepted = 0;

void setup() {
  Serial.begin(9600);

  lcd.init();
  lcd.backlight();

  IrReceiver.begin(PIN_IR, ENABLE_LED_FEEDBACK);

  Wire.begin(SLAVE_ADDRESS);        // with an address = join as SLAVE
  Wire.onReceive(receiveEvent);
  Wire.onRequest(requestEvent);

  // Deliberately NOT writing to the LCD here. Wire.begin() has already
  // registered the I2C callbacks above, so a poll from the master can land
  // inside these writes and crash the board — observed as a reset loop
  // printing "Slave ready." once per second. The first real render happens
  // in loop() once a payload arrives, which is safe.
  // writeLine(0, "BOOTING");
  // writeLine(1, "");

  Serial.println(F("Slave ready."));
}

void loop() {
  handleIR();
  handleDisplay();
}


// I2C CALLBACKS — interrupt context. Touch variables only. No lcd.* here.

void receiveEvent(int howMany) {
  uint8_t i = 0;
  while (Wire.available() && i < 4) {
    rxBuffer[i++] = Wire.read();
  }
  while (Wire.available()) {   // drain anything unexpected so the next
    Wire.read();               // packet starts cleanly aligned
  }
  if (i == 4) {
    newDataReady = true;
  }
}

void requestEvent() {
  Wire.write(pendingCommand);
  pendingCommand = CMD_NONE;   // cleared only once the master has read it
}


// IR HANDLING

void handleIR() {
  if (!IrReceiver.decode()) {
    return;
  }

  bool isRepeat = IrReceiver.decodedIRData.flags & IRDATA_FLAGS_IS_REPEAT;

  if (!isRepeat) {
    uint8_t cmd = mapIRToCommand(IrReceiver.decodedIRData.command);

    if (cmd != CMD_NONE && (millis() - lastIRAccepted >= IR_DEBOUNCE_MS)) {
      if (pendingCommand == CMD_NONE) {
        pendingCommand = cmd;
        lastIRAccepted = millis();
        Serial.print(F("IR command queued: "));
        Serial.println(cmd);
      }
    }
  }

  IrReceiver.resume();
}

uint8_t mapIRToCommand(uint8_t irCode) {
  if (irCode == IR_CODE_ACTIVATE) return CMD_ACTIVATE;
  if (irCode == IR_CODE_TOGGLE)   return CMD_TOGGLE;
  if (irCode == IR_CODE_RESET)    return CMD_RESET;
  return CMD_NONE;   // every other button is ignored
}

// DISPLAY

void handleDisplay() {
  if (!newDataReady) {
    return;
  }

  // Copy out of the volatile buffer with interrupts off, so a poll landing
  // mid-copy cannot hand us half of one packet and half of the next.
  uint8_t state, mode, lightPct, gasPct;
  noInterrupts();
  state    = rxBuffer[0];
  mode     = rxBuffer[1];
  lightPct = rxBuffer[2];
  gasPct   = rxBuffer[3];
  newDataReady = false;
  interrupts();

  // Re-render only on an actual change. Without this the LCD is rewritten
  // every 50 ms, flooding the bus for no visible benefit.
  if (state == lastState && mode == lastMode &&
      lightPct == lastLight && gasPct == lastGas) {
    return;
  }
  lastState = state;
  lastMode  = mode;
  lastLight = lightPct;
  lastGas   = gasPct;

  renderState(state, mode, lightPct, gasPct);
}

void renderState(uint8_t state, uint8_t mode, uint8_t lightPct, uint8_t gasPct) {
  char buf[17];

  switch (state) {
    case STANDBY:
      writeLine(0, "AWAITING RITUAL");
      writeLine(1, "");
      break;

    case ACTIVE_MONITORING:
      writeLine(0, "MONITORING");
      if (mode == 0) {
        snprintf(buf, sizeof(buf), "Light: %u%%", lightPct);
      } else {
        snprintf(buf, sizeof(buf), "Air Purity:%u%%", gasPct);
      }
      writeLine(1, buf);
      break;

    case GAS_ALERT:
      writeLine(0, "TOXIC PURGE");
      writeLine(1, "");
      break;

    case BLACKOUT_ALERT:
      writeLine(0, "NOCTIS PROTOCOL");
      writeLine(1, "");
      break;

    case TEMP_EMERGENCY:
      writeLine(0, "COOKED");
      writeLine(1, "");
      break;

    case MULTI_FAULT:
      writeLine(0, ">1 PROBLEMS");
      writeLine(1, "DETECTED");
      break;

    // Temperature plus at least one other fault. Not a spec state — added
    // deliberately. Worded ">1 & TEMP PROBS" / "DETECTED" (15 and 8 chars)
    // to fit 16x2 and for multi-fault wording above.
    case TEMP_MULTI_FAULT:
      writeLine(0, ">1 & TEMP PROBS");
      writeLine(1, "DETECTED");
      break;

    default:
      writeLine(0, "UNKNOWN STATE");
      writeLine(1, "");
      break;
  }
}

// Writes exactly 16 characters, space-padded.
void writeLine(uint8_t row, const char* text) {
  lcd.setCursor(0, row);
  uint8_t i = 0;
  while (i < 16 && text[i] != '\0') {
    lcd.print(text[i]);
    i++;
  }
  while (i < 16) {
    lcd.print(' ');
    i++;
  }
}
